    <?php

namespace App\Livewire;

use Illuminate\Support\Str;
use Livewire\Component;

class RecoveryCodesPanel extends Component
{
    /** @var list<string> */
    public array $codes = ['4F8K-92QA', 'M3ZP-71LD', '7TQX-0BNE', 'C8WV-5YHR', '2JDA-KE63', 'R9UG-X4MS'];

    public function regenerate(): void
    {
        $this->codes = collect(range(1, 6))
            ->map(fn () => strtoupper(Str::random(4)).'-'.strtoupper(Str::random(4)))
            ->all();
    }

    public function render()
    {
        return view('livewire.recovery-codes-panel');
    }
}

</div>

<!-- once per page: copy (with feedback), download and print -->
<script>
  document.addEventListener('click', async (e) => {
    const btn = e.target.closest('[data-oc-recovery]');
    if (!btn) return;
    const panel = btn.closest('.oc-recovery-codes');
    const codes = Array.from(panel.querySelectorAll('.oc-recovery-codes-list li')).map((li) => li.textContent.trim());
    const text = codes.join('\n');
    const action = btn.dataset.ocRecovery;
    if (action === 'copy') {
      try { await navigator.clipboard.writeText(text); } catch { return; }
      // data-copied-label: feedback on the button and in the live region (.oc-recovery-codes-status)
      const label = btn.querySelector('.oc-recovery-codes-label');
      const status = panel.querySelector('.oc-recovery-codes-status');
      const done = btn.dataset.copiedLabel || 'Copied';
      if (label) { label.dataset.label ??= label.textContent; label.textContent = done; }
      if (status) status.textContent = done;
      btn.classList.add('is-copied');
      clearTimeout(btn.ocCopyTimer);
      btn.ocCopyTimer = setTimeout(() => {
        btn.classList.remove('is-copied');
        if (label) label.textContent = label.dataset.label;
        if (status) status.textContent = '';
      }, 2000);
      return;
    }
    if (action === 'print') {
      // only the codes on paper: a hidden iframe with the title and the list
      const esc = (t) => t.replace(/[&<>"']/g, (ch) => `&#${ch.charCodeAt(0)};`);
      const title = esc(panel.querySelector('.oc-recovery-codes-title')?.textContent.trim() || 'Recovery codes');
      const frame = document.createElement('iframe');
      frame.setAttribute('aria-hidden', 'true');
      frame.style.cssText = 'position:fixed;inline-size:0;block-size:0;border:0;inset:auto auto 0 0';
      frame.srcdoc = `<!doctype html><meta charset="utf-8"><title>${title}</title><style>body{font:14px ui-monospace,Consolas,monospace;margin:2cm}h1{font:600 16px system-ui,sans-serif}ul{list-style:none;padding:0;columns:2}</style><h1>${title}</h1><ul>${codes.map((c) => `<li>${esc(c)}</li>`).join('')}</ul>`;
      frame.addEventListener('load', () => { frame.contentWindow.print(); setTimeout(() => frame.remove(), 1000); });
      document.body.append(frame);
      return;
    }
    const a = document.createElement('a');
    a.href = URL.createObjectURL(new Blob([text + '\n'], { type: 'text/plain' }));
    a.download = btn.dataset.filename || 'recovery-codes.txt';
    a.click();
    URL.revokeObjectURL(a.href);
  });
</script>

    Reference

    Styles live in packages/ui/src/components.css. Colors are theme tokens (--oc-*), so light and dark mode work without extra classes.

    Class / attributeDescription
    .oc-recovery-codesPanel (warning tint)
    .oc-recovery-codes-title / -warningHeading and explanation
    .oc-recovery-codes-listTwo-column monospace grid (selectable with one click)
    .oc-recovery-codes-actionsButton row. data-oc-recovery="copy|download|print" wires a button to the optional script.
    .oc-recovery-codes-warning iconAn svg before the text (mdi:alert-outline) takes the warning color (0.17.0).
    data-copied-label + .oc-recovery-codes-labelThe copy button shows the copied label (default "Copied") for 2s; .oc-recovery-codes-icon-copy / -copied swap on .is-copied (0.17.0). Prop copiedLabel.
    .oc-recovery-codes-statusVisually hidden role="status" live region that announces the copied label (0.17.0).
    print, printLabel, icons (props)A Print button that prints only the codes; icons on the buttons and the warning (default on) (0.17.0). React also exports printRecoveryCodes(codes, title).
