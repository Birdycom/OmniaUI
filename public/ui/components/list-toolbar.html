    List toolbar

    The bar above a list: a search field, the result count, a cards / table view toggle and action buttons. The toggle is a radio group; inside .oc-list-toolbar-scope it switches the visible view with :has(), no JavaScript. All components.

    Search, count and view toggle

    Click the two view icons. Each panel has data-view="cards" or data-view="table"; the checked radio's value decides which one shows. Search and filtering stay your app's job. Snippet: packages/ui/snippets/list-toolbar.html.

    import { useState } from "react";
import { Button, Card, CardContent, Input, ListToolbar, Table, Webicon } from "@birdycom/omniaui-react";

const products = ["Oak table", "Linen chair", "Brass lamp"];
const views = [
  { value: "cards", label: "Cards", icon: <Webicon name="mdi:view-grid-outline" /> },
  { value: "table", label: "Table", icon: <Webicon name="mdi:table" /> },
];

export function Example() {
  const [query, setQuery] = useState("");
  const [view, setView] = useState("cards");
  const shown = products.filter((p) => p.toLowerCase().includes(query.toLowerCase()));
  return (
    <div style={{ width: "100%" }}>
      <ListToolbar
        aria-label="Products"
        search={<Input type="search" placeholder="Search products" aria-label="Search products" value={query} onChange={(e) => setQuery(e.target.value)} />}
        count={`${shown.length} items`}
        views={views}
        view={view}
        onViewChange={setView}
        actions={<Button size="sm">New product</Button>}
      />
      {view === "cards" ? (
        <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill,minmax(10rem,1fr))", gap: ".75rem" }}>
          {shown.map((p) => <Card key={p} className="oc-card--sm"><CardContent>{p}</CardContent></Card>)}
        </div>
      ) : (
        <Table>
          <thead><tr><th>Product</th></tr></thead>
          <tbody>{shown.map((p) => <tr key={p}><td>{p}</td></tr>)}</tbody>
        </Table>
      )}
    </div>
  );
}

    <script setup lang="ts">
import { computed, ref } from "vue";
import { OcButton, OcCard, OcInput, OcListToolbar } from "@birdycom/omniaui-vue";

const products = ["Oak table", "Linen chair", "Brass lamp"];
const query = ref("");
const view = ref("cards");
const shown = computed(() => products.filter((p) => p.toLowerCase().includes(query.value.toLowerCase())));
</script>

<template>
  <div style="width:100%">
    <OcListToolbar
      v-model:view="view"
      aria-label="Products"
      :count="`${shown.length} items`"
      :views="[{ value: 'cards', label: 'Cards' }, { value: 'table', label: 'Table' }]"
    >
      <template #search><OcInput v-model="query" type="search" placeholder="Search products" aria-label="Search products" /></template>
      <template #actions><OcButton size="sm">New product</OcButton></template>
    </OcListToolbar>
    <div v-if="view === 'cards'" style="display:grid;grid-template-columns:repeat(auto-fill,minmax(10rem,1fr));gap:.75rem">
      <OcCard v-for="p in shown" :key="p" size="sm">{{ p }}</OcCard>
    </div>
    <table v-else class="oc-table">
      <thead><tr><th>Product</th></tr></thead>
      <tbody><tr v-for="p in shown" :key="p"><td>{{ p }}</td></tr></tbody>
    </table>
  </div>
</template>

    <?php

namespace App\Livewire;

use Livewire\Component;

class ProductList extends Component
{
    public string $search = '';

    public array $products = ['Oak table', 'Linen chair', 'Brass lamp'];

    public function create(): void
    {
        // $this->redirect(route('products.create'));
    }

    public function render()
    {
        $shown = array_values(array_filter($this->products, fn ($p) => str_contains(strtolower($p), strtolower($this->search))));

        return view('livewire.product-list', ['shown' => $shown]);
    }
}

    Reference

    Slots: search, count, views, actions, panels. Colors are theme tokens.

    Class / attributeDescription
    .oc-list-toolbarToolbar row (wraps). Modifiers: --sticky, --bordered
    .oc-list-toolbar-searchSearch area with an .oc-input
    .oc-list-toolbar-count / -spacer / -actionsCount, flexible gap, end buttons
    .oc-list-toolbar-viewsRadio group frame
    .oc-list-toolbar-viewlabel + radio (value cards / table / list) + icon
    .oc-list-toolbar-scope + .oc-list-toolbar-panel[data-view]CSS-only view switch
