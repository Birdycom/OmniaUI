    Chip filter

    Rounded filter chips above a list or grid. Use checkboxes to pick several filters, radios to pick one. :has(input:checked) highlights the chip, and the inputs submit with the form. No JavaScript. All components.

    Demo

    snippets/chip-filter.html 0 KB JavaScriptBaselinev0.6.0

    Plain HTML and CSS. The markup is also available as the snippet packages/ui/snippets/chip-filter.html.

    Click the chips: each one is a checkbox, so several can be on at once. Documents is disabled because it has no items.

    import { useState } from "react";

const options = [
  { value: "video", label: "Video", count: 24 },
  { value: "audio", label: "Audio", count: 8 },
  { value: "image", label: "Images", count: 112 },
  { value: "docs", label: "Documents", count: 0, disabled: true },
];

export function Example() {
  const [types, setTypes] = useState<string | string[]>(["video", "image"]);
  return <ChipFilter multiple name="type" aria-label="Filter by type" options={options} value={types} onChange={setTypes} />;
}

    <?php

namespace App\Livewire;

use Livewire\Component;

class TypeFilter extends Component
{
    public array $types = ['video', 'image'];

    public array $options = [
        'video' => ['Video', 24],
        'audio' => ['Audio', 8],
        'image' => ['Images', 112],
        'docs' => ['label' => 'Documents', 'count' => 0, 'disabled' => true],
    ];

    public function toggle(string $type): void
    {
        $this->types = in_array($type, $this->types, true)
            ? array_values(array_diff($this->types, [$type]))
            : [...$this->types, $type];
    }

    public function render()
    {
        return view('livewire.type-filter');
    }
}

{{-- resources/views/livewire/type-filter.blade.php --}}
<div>
    {{-- attributes go to the group: the change of a chip bubbles up and toggles its value in $types --}}
    <x-oc::chip-filter name="type" multiple label="Filter by type" :options="$options" :value="$types" wire:change="toggle($event.target.value)" />
</div>

    One choice, small, scrolling v0.17.0

    Radios give a single choice (arrow keys move between chips). --sm makes them compact; --scroll keeps them on one line that scrolls sideways on small screens. Click a chip or use the arrow keys: only one stays on. The row is narrower than the chips, so scroll it sideways to reach Scheduled.

    import { useState } from "react";
import { ChipFilter } from "@birdycom/omniaui-react";

const options = [
  { value: "quick", label: "Quick" },
  { value: "full", label: "Full scan" },
  { value: "deep", label: "Deep scan" },
  { value: "custom", label: "Custom rules" },

export function Example() {
  const [mode, setMode] = useState<string | string[]>("quick");

      scroll
      name="mode"
      aria-label="Scan mode"
      options={options}
      value={mode}
      onChange={setMode}
      style={{ maxInlineSize: "20rem" }}
    />
  );
}

    Reference

    Styles live in packages/ui/src/parts/forms-chip-filter.css. For a joined pill switcher, see Segmented control.

    Class / attributeDescription
    .oc-chip-filterGroup (<fieldset>, role="group" or "radiogroup"). Modifiers: --scroll, --sm.
    .oc-chip-filter-itemPill label around one checkbox or radio
    .oc-chip-filter-countOptional count bubble; primary when the chip is checked
