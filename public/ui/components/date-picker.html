    Date picker

    A button that opens a calendar in a popover and shows the chosen date. Zero-JavaScript alternative: a native date input with the themed .oc-input. The popover gives top layer, light dismiss and Esc; the date logic is yours. The demo includes a small script that renders the month, steps months, picks a day and handles the calendar keys. All components.

    Baseline

    HTML and CSS, plus a small script that runs the calendar. The markup is also available as the snippet packages/ui/snippets/date-picker.html.

<script>
// Date picker demo: renders the month in the popover, steps months, picks a day (updates the trigger, closes the popover)
// and handles the calendar keys. Fires "oc-datepicker-change" ({ value: "YYYY-MM-DD" }) on the trigger.
(() => {
  const pad = (n) => String(n).padStart(2, "0");
  const iso = (d) => `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
  const parse = (s) => { const [y, m, d] = s.split("-").map(Number); return new Date(y, m - 1, d); };
  const addDays = (d, n) => new Date(d.getFullYear(), d.getMonth(), d.getDate() + n);
  const addMonths = (d, n) => {
    const t = new Date(d.getFullYear(), d.getMonth() + n, 1);
    return new Date(t.getFullYear(), t.getMonth(), Math.min(d.getDate(), new Date(t.getFullYear(), t.getMonth() + 1, 0).getDate()));
  };
  const long = new Intl.DateTimeFormat("en-GB", { day: "numeric", month: "long", year: "numeric" });
  const month = new Intl.DateTimeFormat("en-GB", { month: "long", year: "numeric" });

  for (const trigger of document.querySelectorAll(".oc-date-picker-trigger[popovertarget]")) {
    const pop = document.getElementById(trigger.getAttribute("popovertarget"));
    const cal = pop && pop.querySelector(".oc-calendar");
    if (!cal) continue;
    const label = trigger.querySelector("span");
    const title = cal.querySelector(".oc-calendar-title");
    const grid = cal.querySelector(".oc-calendar-grid");
    const body = grid.tBodies[0];
    const [prev, next] = cal.querySelectorAll(".oc-calendar-nav");
    const initial = body.querySelector('[aria-selected="true"]');
    let selected = initial ? parse(initial.dataset.date) : null;
    let active = selected || new Date();

    const render = (focus) => {
      const today = iso(new Date());
      const first = new Date(active.getFullYear(), active.getMonth(), 1);
      let day = addDays(first, -((first.getDay() + 6) % 7));
      const rows = [];
      for (let w = 0; w < 6; w++) {
        const cells = [];
        for (let i = 0; i < 7; i++, day = addDays(day, 1)) {
          const key = iso(day);
          const cls = ["oc-calendar-day"];
          if (day.getMonth() !== active.getMonth()) cls.push("oc-calendar-day--outside");
          if (key === today) cls.push("oc-calendar-day--today");
          cells.push(`<td><button class="${cls.join(" ")}" type="button" data-date="${key}" tabindex="${key === iso(active) ? 0 : -1}"`
            + `${key === today ? ' aria-current="date"' : ""}${selected && key === iso(selected) ? ' aria-selected="true"' : ""}`
            + ` aria-label="${long.format(day)}">${day.getDate()}</button></td>`);
        }
        rows.push(`<tr>${cells.join("")}</tr>`);
      }
      body.innerHTML = rows.join("");
      const name = month.format(active);
      title.textContent = name;
      cal.setAttribute("aria-label", name);
      grid.setAttribute("aria-label", name);
      if (focus) body.querySelector('[tabindex="0"]').focus();
    };

    const pick = (date) => {
      selected = active = date;
      label.textContent = long.format(date);
      trigger.removeAttribute("data-empty");
      render(false);
      pop.hidePopover();
      trigger.focus();
      trigger.dispatchEvent(new CustomEvent("oc-datepicker-change", { bubbles: true, detail: { value: iso(date) } }));
    };

    pop.addEventListener("toggle", (e) => {
      if (e.newState !== "open") return;
      active = selected || new Date();
      render(true);
    });
    prev.addEventListener("click", () => { active = addMonths(active, -1); render(false); });
    next.addEventListener("click", () => { active = addMonths(active, 1); render(false); });
    body.addEventListener("click", (e) => {
      const btn = e.target.closest(".oc-calendar-day");
      if (btn) pick(parse(btn.dataset.date));
    });
    body.addEventListener("keydown", (e) => {
      const steps = {
        ArrowLeft: (d) => addDays(d, -1), ArrowRight: (d) => addDays(d, 1),
        ArrowUp: (d) => addDays(d, -7), ArrowDown: (d) => addDays(d, 7),
        PageUp: (d) => addMonths(d, -1), PageDown: (d) => addMonths(d, 1),
        Home: (d) => addDays(d, -((d.getDay() + 6) % 7)), End: (d) => addDays(d, 6 - ((d.getDay() + 6) % 7)),
      };
      if (!steps[e.key]) return;
      e.preventDefault();
      active = steps[e.key](active);
      render(true);
    });
  }
})();
</script>

    Reference

    Styles live in packages/ui/src/components-extra.css. Colors are theme tokens (--oc-*), so light and dark mode work without extra classes.

    Platform features used

    CSS featureRole
    color-schemeThemes the native picker
    position-areaPopover opens below the trigger

    Anatomy

    • Trigger an outline button; show the selected date (or a placeholder with data-empty).
    • Panel the popover component with --calendar, so the calendar is the panel.
    • Native option input[type=date] uses the platform picker; it inherits color-scheme, so text and picker follow the page theme.

    Accessibility and keyboard

    Accessibility

    • The popover content is a dialog-like widget: set aria-haspopup="dialog" on the trigger and label the calendar.
    • Always offer typing a date (an input) for users who do not want to pick from a grid.

    Keyboard

    • Enter opens, Esc closes and returns focus; calendar keys as on the calendar page.
