    Phone input

    A country dial code (native <select>) and the national number (type="tel") inside one frame. Mobile keyboards show digits, autofill fills the number, and the browser checks the pattern. The React and Vue wrappers add digit grouping and return one value like +32470123456. All components.

    Demo

    snippets/phone-input.html 0 KB JavaScriptBaselinev0.6.0

    Plain HTML and CSS. The markup is also available as the snippet packages/ui/snippets/phone-input.html.

    Pick a country code and type the number. The code and the number are one control with one frame.

    <?php

namespace App\Livewire;

use Livewire\Component;

class MobileNumber extends Component
{
    public string $dial = '+32';

    public string $phone = '';

    public function save(): void
    {
        $this->validate([
            'dial' => ['required', 'in:+32,+31,+33,+49'],
            'phone' => ['required', 'regex:/^[0-9 ]{8,14}$/'],
        ]);
        // store the full number: $this->dial.' '.$this->phone
    }

    public function render()
    {
        return view('livewire.mobile-number');
    }
}

{{-- resources/views/livewire/mobile-number.blade.php --}}
<form wire:submit="save">
    {{-- dial-model binds the country code (wire:model on the select); wire:model binds the number --}}
    <x-oc::form-field label="Mobile number" for="phone" name="phone" description="Digits only; spaces are fine." style="max-inline-size:22rem">
        <x-oc::phone-input id="phone" dial-model="dial" wire:model.blur="phone" :error="$errors->has('phone')" placeholder="470 12 34 56" />
    </x-oc::form-field>
</form>

    <?php

namespace App\Livewire;

use Livewire\Component;

class PhoneCheck extends Component
{
    public string $phone = '12';

    public function mount(): void
    {
        $this->validateOnly('phone');
    }

    public function updatedPhone(): void
    {
        $this->validateOnly('phone');
    }

    protected function rules(): array
    {
        return ['phone' => ['required', 'min:8']];
    }

    protected function messages(): array
    {
        return ['phone.min' => 'This number is too short.'];
    }

    public function render()
    {
        return view('livewire.phone-check');
    }
}

{{-- resources/views/livewire/phone-check.blade.php --}}
<div>
    <x-oc::form-field label="Phone" for="phone" name="phone" style="max-inline-size:22rem">
        <x-oc::phone-input id="phone" dial="+31" :countries="['+31' => 'NL +31']" wire:model.blur="phone" :error="$errors->has('phone')" />
    </x-oc::form-field>
    <div style="max-inline-size:16rem">
        <x-oc::phone-input size="sm" :countries="['+32' => 'BE +32']" placeholder="470 12 34 56" aria-label="Phone (small)" />
    </div>
</div>

    Reference

    Styles live in packages/ui/src/parts/forms-phone-input.css. Colors are theme tokens (--oc-*), so light and dark mode work without extra classes.

    Class / attributeDescription
    .oc-phone-inputFrame around both parts. Modifiers: --sm, --error.
    .oc-phone-input-dialNative select with the dial codes
    .oc-phone-input-numberThe national number (type="tel", autocomplete="tel-national")
    aria-invalid="true" / :user-invalidRed frame on the whole control

    Framework wrappers
