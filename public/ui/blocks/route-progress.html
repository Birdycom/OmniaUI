    Route progress

    What a visitor or a team sees while walking a route, a tour or a course: how far they are, the next step with its distance, and every step on one line with its state (done, wrong, current, to do, locked). It needs no JavaScript: your app sets the states and the numbers. Install with npx omniaui add route-progress. All blocks.

    Demo

    blocks/route-progress.html0 KB JavaScriptBaseline

    Plain HTML and CSS: every step and the next-step card are buttons your app connects to its own screens.

    Route progress v0.16.0

    Look at the markers on the route line: green with points for a step that is done, grey for a wrong answer, a blue ring for the step to go to now and a dashed circle for the finish that is still closed. Hover a row or press Tab to move through the steps.

    <?php

namespace App\Livewire;

use Livewire\Component;

class TeamRoute extends Component
{
    public string $route = 'Route 1';

    public string $team = 'Team 1';

    public ?int $opened = null;

    /** @var list<array<string, mixed>> */
    public array $steps = [
        ['name' => 'Old gate', 'sub' => 'Right answer', 'value' => '+20', 'state' => 'done', 'marker' => 1],
        ['name' => 'Market square', 'sub' => 'Wrong answer', 'value' => '0', 'state' => 'wrong', 'marker' => 2],
        ['name' => 'Team photo', 'sub' => 'Photo sent', 'value' => '+20', 'state' => 'done', 'marker' => 'mdi:camera-outline', 'markerLabel' => 'Photo task'],
        ['name' => 'Town hall', 'sub' => 'Market street 1', 'value' => '240 m', 'state' => 'current', 'marker' => 3],
        ['name' => 'Finish', 'sub' => 'Opens after 4 steps', 'value' => '1.1 km', 'state' => 'locked', 'marker' => 'mdi:flag-checkered', 'markerLabel' => 'Finish'],
    ];

    public function open(int|string $index): void
    {
        $this->opened = (int) $index;
    }

    public function render()
    {
        return view('livewire.team-route');
    }
}

    <x-oc::block.route-progress>

class RouteFinish extends Component
{
    public int $score = 80;

    /** @var list<array<string, mixed>> */
    public array $steps = [
        ['name' => 'Old gate', 'sub' => 'Right answer', 'value' => '+20', 'state' => 'done', 'marker' => 1],
        ['name' => 'Market square', 'sub' => 'Wrong answer', 'value' => '0', 'state' => 'wrong', 'marker' => 2],
        ['name' => 'Team photo', 'sub' => 'Photo sent', 'value' => '+20', 'state' => 'done', 'marker' => 'mdi:camera-outline', 'markerLabel' => 'Photo task'],
        ['name' => 'Town hall', 'sub' => 'Right answer', 'value' => '+20', 'state' => 'done', 'marker' => 3],
        ['name' => 'River bridge', 'sub' => 'Right answer', 'value' => '+20', 'state' => 'done', 'marker' => 4],
        ['name' => 'Finish', 'sub' => 'Open now', 'value' => '350 m', 'state' => 'current', 'marker' => 'mdi:flag-checkered', 'markerLabel' => 'Finish'],
    ];

    public function checkIn(): void
    {
        $this->redirect('/route/finish', navigate: true);
    }

    public function done(): int
    {
        return collect($this->steps)->whereIn('state', ['done', 'wrong'])->count();
    }

    public function render()
    {
        return view('livewire.route-finish');
    }
}
