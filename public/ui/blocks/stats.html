    <?php

namespace App\Livewire;

use Livewire\Component;

class Stats extends Component
{
    /** Label => figure; fill it from your own data in mount(). */
    public array $stats = [
        'Years in business' => '12',
        'Customers' => '140',
        'Apps in production' => '35',
        'Average response time' => '4 h',
    ];

    public function render()
    {
        return view('livewire.stats');
    }
}

{{-- resources/views/livewire/stats.blade.php --}}
<x-oc::block.stats>
    @foreach ($stats as $label => $value)
        <div class="oc-block-stat">
            <div class="oc-block-stat-value">{{ $value }}</div>
            <div class="oc-block-stat-label">{{ $label }}</div>
        </div>
    @endforeach
</x-oc::block.stats>

    Styles live in packages/blocks/src/blocks.css. Colors are theme tokens, so everything follows light and dark mode.

    .oc-block-stat-valueLarge accent-colored metric
    .oc-block-stat-labelMuted label below value
