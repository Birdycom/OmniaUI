    <?php

namespace App\Livewire;

use Livewire\Attributes\Validate;
use Livewire\Component;

class Register extends Component
{
    #[Validate('required|string|max:255')]
    public string $name = '';

    #[Validate('required|email|max:255')]
    public string $email = '';

    #[Validate('required|min:8|confirmed')]
    public string $password = '';

    public string $password_confirmation = '';

    public function register(): void
    {
        $this->validate();
        // create the user, log in and redirect
    }

    public function render()
    {
        return view('livewire.register');
    }
}

{{-- resources/views/livewire/register.blade.php --}}
<div>
    {{-- form takes wire:submit; the fields and footer slots replace those regions, the rest comes from the block --}}
    <x-oc::block.signup-01 heading-level="2" :form="['wire:submit' => 'register']">
        <x-slot:fields>
            <x-oc::form-field label="Full name" for="name" name="name"><x-oc::wire.input id="name" model="name" autocomplete="name" required placeholder="Contact 1" /></x-oc::form-field>
            <x-oc::form-field label="E-mail" for="email" name="email"><x-oc::wire.input id="email" model="email" blur type="email" autocomplete="email" required placeholder="name@company1.example" /></x-oc::form-field>
            <x-oc::password-input label="Password" id="password" wire:model="password" autocomplete="new-password" :error="$errors->has('password')" required />
            @error('password') <p class="oc-field-error">{{ $message }}</p> @enderror
            <x-oc::password-input label="Confirm password" id="password_confirmation" wire:model="password_confirmation" autocomplete="new-password" required />
            <p class="oc-block-authcard-hint">Must be at least 8 characters long.</p>
            <x-oc::button type="submit" class="oc-block-authcard-submit">Create account</x-oc::button>
        </x-slot:fields>
        <x-slot:footer>Already have an account? <a href="/login" wire:navigate>Sign in</a></x-slot:footer>
    </x-oc::block.signup-01>
</div>

    About OmniaUI Pro

    Reference

    Styles live in packages/blocks/src/blocks.css. Colors are theme tokens (--oc-*), so light and dark mode work without extra classes. Interactions are native HTML and CSS (details, popover, dialog, :has()).

    Class / attributeDescription
    .oc-block-authpagePage wrapper that centers one card. Modifier: --plain (page background instead of muted)
    .oc-block-authcardCard column. Modifiers: --flat (no card chrome), --wide
    .oc-block-authcard-logoIcon tile above the title
    .oc-block-authcard-head / -title / -textCentered heading block, title and description
    .oc-block-authcard-fields / -rowField stack and a justified row (remember me / forgot password)
    .oc-block-authcard-submitFull-width submit button
    .oc-block-authcard-foot / -legal / -hintSwitch-form line, legal line, small helper text
    .oc-block-authcard-centerCentered column (OTP input)
    .oc-block-authsocialAuto-fit grid of social buttons
    .oc-block-authduoSplit layout root (container query: two columns from 42rem of its own width). Modifier: --reverse
    .oc-block-authduo-grid / -panel / -formGrid, brand panel (modifiers --image, --muted) and form column
    .oc-block-authduo-logo / -quotePanel logo and quote
    .oc-block-authminMinimal centered column without a card
    .oc-separator-textDivider with centered text (separator component)
