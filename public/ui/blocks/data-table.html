    Data table

    The list page pattern: a card that holds a toolbar (search, status pills), a sortable table with selection and row actions, and a footer with the entry count, page size and pages. With data-table.js it works in the browser (try search, the pills, sorting, selecting and deleting); server side (Livewire) the same look renders links instead. Type tabs, filter selects, rich name cells and cards: the Data table component. All blocks.

import { DataTable, type DataTableColumn, type DataTableQuery, type DataTableRow } from "@birdycom/omniaui-react";

const columns: DataTableColumn[] = [
  { key: "name", header: "Name", hideable: false },
  { key: "sku", header: "Article ID" },
  { key: "price", header: "Price", numeric: true },
  { key: "supplier", header: "Supplier" },
];

// One fetch for every change: onQueryChange gets the whole query (sort, page, page size, search, trash, filters).
export function Example() {
  const [query, setQuery] = useState<Partial<DataTableQuery>>({ page: 1, pageSize: 10, trashed: "active" });
  const [data, setData] = useState<{ rows: DataTableRow[]; total: number }>({ rows: [], total: 0 });
  const [loading, setLoading] = useState(true);
  const [visible, setVisible] = useState(["name", "sku", "price"]);

  useEffect(() => {
    let current = true;
    setLoading(true);
    fetch("/api/articles", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(query) })
      .then((res) => res.json())
      .then((json: { rows: DataTableRow[]; total: number }) => {
        if (!current) return; // a newer query replaced this one
        setData(json);
        setLoading(false);
      });
    return () => {
      current = false;
    };
  }, [query]);

  return (
    <DataTable
      title="Articles"
      server
      columns={columns}
      rows={data.rows}
      total={data.total}
      loading={loading}
      page={query.page}
      pageSize={10}
      archive
      visibleColumns={visible}
      onVisibleColumnsChange={setVisible}
      onQueryChange={setQuery}
    />
  );
}

use Illuminate\Database\Query\Builder;
use Illuminate\Support\Facades\DB;
use Omniacss\Livewire\DataTable;

// Livewire is server-driven already: search, sort and pages run in the query. 0.17.0 adds the trash pills
// ($trashed, ?trashed= in the URL), the column chooser ($hidden) and skeleton rows while a request runs.
class ArticlesServerTable extends DataTable
{
    protected function title(): ?string
    {
        return 'Articles';
    }

    protected function columns(): array
    {
        return [
            'name' => ['label' => 'Name', 'hideable' => false],
            'sku' => 'Article ID',
            'price' => ['label' => 'Price', 'numeric' => true, 'searchable' => false],
            'supplier' => 'Supplier',
        ];
    }

    protected function query(): Builder
    {
        return DB::table('articles');
    }

    // A model with SoftDeletes gets active / including trash / trash only by itself. Here: an archive status column.
    protected function trashedOptions(): array
    {
        return ['active', 'all', 'archived'];
    }

    protected function applyTrashed(EloquentBuilder|Builder $query): EloquentBuilder|Builder
    {
        return match ($this->trashed) {
            'archived' => $query->where('status', 'Archived'),
            'all' => $query,
            default => $query->where('status', '!=', 'Archived'),
        };
    }

    protected function columnChooser(): bool
    {
        return true;
    }
}
