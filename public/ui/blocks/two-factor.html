    <?php

namespace App\Livewire;

use Livewire\Component;

class TwoFactorSetup extends Component
{
    public string $setupKey = 'JBSWY3DPEHPK3PXP';

    public string $code = '';

    public bool $enabled = false;

    /** @var list<string> */
    public array $recoveryCodes = ['4F8K-92QA', 'M3ZP-71LD', '7TQX-0BNE', 'C8WV-5YHR'];

    public function enable(): void
    {
        $this->validate(['code' => 'required|digits:6']);
        // verify the code against $setupKey, then store the secret
        $this->enabled = true;
    }

    public function render()
    {
        return view('livewire.two-factor-setup');
    }
}

{{-- resources/views/livewire/two-factor-setup.blade.php --}}
<form wire:submit="enable">
    <x-oc::block.two-factor>
        <x-slot:qr><img src="/user/two-factor-qr.svg" alt="QR code"></x-slot:qr>
        <x-slot:key><x-oc::copy-field :value="$setupKey" label="Setup key" /></x-slot:key>
        <x-slot:code><x-oc::wire.otp model="code" :length="6" complete="enable" label="Authentication code" /></x-slot:code>
        <x-slot:recovery><x-oc::recovery-codes :codes="$recoveryCodes" /></x-slot:recovery>
    </x-oc::block.two-factor>
</form>

    Styles live in packages/blocks/src/blocks.css. Colors are theme tokens (--oc-*), so light and dark mode work without extra classes. Slots: QR code, setup key, code input, recovery codes.

    .oc-block-twofactor-stepOne step (heading + content)
    .oc-block-twofactor-centerCentered column for the QR code
    .oc-block-twofactor-qrQR placeholder box (replace with your img/svg)
