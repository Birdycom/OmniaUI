Popover

Free-form content in the top layer, tethered to its trigger. For lists of actions use the dropdown component. All components.

Demo

Plain HTML and CSS. Below: the markup that builds it, also available as the snippet packages/ui/snippets/popover.html.

Click Dimensions: a small panel with fields opens next to the button. Escape or a click outside closes it.

Dimensions

Set the width and height of the layer.

import { useState } from "react"; import { FormField, Input, Popover } from "@birdycom/omniaui-react"; export function Example() { const [width, setWidth] = useState("100%"); return ( <Popover trigger="Dimensions" title="Dimensions" text="Set the width and height of the layer."> <FormField label="Width" style={{ marginTop: ".75rem" }}> <Input value={width} onChange={(e) => setWidth(e.target.value)} /> </FormField> </Popover> ); } <script setup lang="ts"> import { ref } from 'vue' import { OcFormField, OcInput, OcPopover } from '@birdycom/omniaui-vue' const width = ref('100%') </script> <template> <OcPopover label="Dimensions" title="Dimensions" text="Set the width and height of the layer."> <OcFormField label="Width" style="margin-top:.75rem" v-slot="f"> <OcInput :id="f.id" v-model="width" /> </OcFormField> </OcPopover> </template> <x-oc::popover id="oc-popover-demo" trigger="Dimensions" title="Dimensions" text="Set the width and height of the layer."> <x-oc::form-field label="Width" for="oc-popover-width" style="margin-top:.75rem"> <x-oc::input id="oc-popover-width" name="width" value="100%" /> </x-oc::form-field> </x-oc::popover> <?php namespace App\Livewire; use Livewire\Component; class LayerDimensions extends Component { public string $width = '100%'; public function render() { return view('livewire.layer-dimensions'); } } {{-- resources/views/livewire/layer-dimensions.blade.php --}} <div> {{-- wire:ignore.self keeps the popover open while Livewire updates the field --}} <x-oc::popover id="oc-popover-demo" wire:ignore.self trigger="Dimensions" title="Dimensions" text="Set the width and height of the layer."> <x-oc::form-field label="Width" for="oc-popover-width" style="margin-top:.75rem"> <x-oc::wire.input model="width" blur id="oc-popover-width" /> </x-oc::form-field> </x-oc::popover> </div>
