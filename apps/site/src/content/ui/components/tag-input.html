Tag input

Chips (.oc-tag) can be used alone as labels. .oc-tag-input wraps chips and a text field in one bordered control. Adding on Enter and removing are handled by the wrapper component. All components.

Demo

Plain HTML and CSS. Below: the markup that builds it, also available as the snippet packages/ui/snippets/tag-input.html. Type a tag and press Enter or a comma, click × to remove one, or pick a suggestion.

Reference

Styles live in packages/ui/src/components.css. Colors are theme tokens (--oc-*), so light and dark mode work without extra classes.

Optional script

Plain HTML pages need this small listener (React, Vue and Livewire wrappers have the behaviour built in). One script handles every tag input on the page.

<script> // Tag input: Enter or comma adds a tag, × removes one, Backspace in the empty field removes the last one, // a suggestion button adds its text. With data-name="tags[]" on .oc-tag-input every tag carries a hidden field. (() => { const field = (box) => box.querySelector("input:not([type=hidden])"); const tags = (box) => [...box.querySelectorAll(".oc-tag")]; const label = (tag) => tag.firstChild.textContent.trim(); const addTag = (box, text) => { const value = text.trim().replace(/,+$/, ""); if (!value || tags(box).some((t) => label(t) === value)) return; const tag = document.createElement("span"); tag.className = "oc-tag"; tag.append(`${value} `); const remove = document.createElement("button"); remove.type = "button"; remove.className = "oc-tag-remove"; remove.setAttribute("aria-label", `Remove ${value}`); remove.textContent = "×"; tag.append(remove); if (box.dataset.name) { const hidden = document.createElement("input"); hidden.type = "hidden"; hidden.name = box.dataset.name; hidden.value = value; tag.append(hidden); } box.insertBefore(tag, field(box)); }; document.addEventListener("keydown", (e) => { const box = e.target.closest?.(".oc-tag-input"); if (!box || e.target !== field(box)) return; if (e.key === "Enter" || e.key === ",") { e.preventDefault(); addTag(box, e.target.value); e.target.value = ""; } else if (e.key === "Backspace" && !e.target.value) tags(box).pop()?.remove(); }); document.addEventListener("click", (e) => { const remove = e.target.closest?.(".oc-tag-remove"); if (remove) { const box = remove.closest(".oc-tag-input"); remove.closest(".oc-tag").remove(); field(box)?.focus(); return; } const suggestion = e.target.closest?.(".oc-tag-suggestions button"); const box = suggestion?.closest(".oc-tag-suggestions").previousElementSibling; if (box?.matches(".oc-tag-input")) addTag(box, suggestion.textContent); }); })(); </script>
